// ---- DUMMY DATA (same shape as nexman_chart_data.json returned by your API) ----

    years:{ label:"Yearly",

      active_users:[4800,9200,15400,22600,31500],
      bonus_used:[1200,2600,4800,7400,10900],
      staking_done:[520,1150,2200,3600,5200],
      earning_coin:[12000,28000,52000,84000,121000],

  }
};

const C = {blue:'#2563eb', green:'#22c55e', amber:'#f59e0b', purple:'#8b5cf6', grey:'#94a3b8'};
const fmt = n => n.toLocaleString();
const sum = a => a.reduce((x,y)=>x+y,0);

const KPI_DEFS = [
  {key:'active_users',    label:'Active Users',    color:C.grey,   coin:false},
  {key:'bonus_used',      label:'Bonus Used',      color:C.amber,  coin:true},
  {key:'staking_done',    label:'Staking Done',    color:C.purple, coin:false},
  {key:'earning_coin',    label:'Earning Coin',    color:C.blue,   coin:true},
  {key:'coin_withdrawal', label:'Coin Withdrawal', color:C.green,  coin:true},
];

function buildDatasets(p){
  return [
    // grouped bars: blue = earning (in), green = withdrawal (out)
    {type:'bar', label:'Earning Coin', data:p.earning_coin,
      backgroundColor:'rgba(37,99,235,.85)', borderRadius:6, order:3, yAxisID:'y'},
    {type:'bar', label:'Coin Withdrawal', data:p.coin_withdrawal,
      backgroundColor:'rgba(34,197,94,.85)', borderRadius:6, order:3, yAxisID:'y'},
    // trend lines
    {type:'line', label:'Bonus Used', data:p.bonus_used,
      borderColor:C.amber, backgroundColor:C.amber, tension:.4, borderWidth:3,
      pointRadius:3, pointBackgroundColor:C.amber, order:1, yAxisID:'y'},
    {type:'line', label:'Staking Done', data:p.staking_done, hidden:true,
      borderColor:C.purple, backgroundColor:C.purple, tension:.4, borderWidth:2,
      pointRadius:2, order:1, yAxisID:'y1'},
    {type:'line', label:'Active Users', data:p.active_users, hidden:true,
      borderColor:C.grey, backgroundColor:C.grey, tension:.4, borderWidth:2,
      borderDash:[5,4], pointRadius:2, order:1, yAxisID:'y1'},
  ];
}

let chart;
function render(periodKey){
  const p = DATA.periods[periodKey];
  document.getElementById('periodName').textContent = p.label;

  // KPI cards (period totals)
  document.getElementById('kpis').innerHTML = KPI_DEFS.map(d=>{
    const total = sum(p[d.key]);
    const val = d.coin ? fmt(total)+' <span style="font-size:12px;color:var(--muted)">BMAN</span>' : fmt(total);
    return `<div class="kpi"><div class="k"><span class="dot" style="background:${d.color}"></span>${d.label}</div>
            <div class="v">${val}</div></div>`;
  }).join('');

  const cfg = {
    data:{labels:p.labels, datasets:buildDatasets(p)},
    options:{
      responsive:true, maintainAspectRatio:false,
      interaction:{mode:'index', intersect:false},
      plugins:{
        legend:{position:'top', labels:{usePointStyle:true, boxWidth:8, padding:16, font:{size:12}}},
        tooltip:{callbacks:{label:c=>` ${c.dataset.label}: ${fmt(c.parsed.y)}`}}
      },
      scales:{
        x:{grid:{display:false}, ticks:{font:{size:11}}},
        y:{beginAtZero:true, position:'left', grid:{color:'#eef2f7'},
           ticks:{callback:v=>v>=1000?(v/1000)+'K':v}},
        y1:{beginAtZero:true, position:'right', grid:{display:false},
            ticks:{callback:v=>v>=1000?(v/1000)+'K':v}}
      }
    }
  };

  if(chart){ chart.data=cfg.data; chart.options=cfg.options; chart.update(); }
  else{ chart = new Chart(document.getElementById('trendChart'), cfg); }
}

document.getElementById('periodSelect').addEventListener('change', e=>render(e.target.value));
render('months');
